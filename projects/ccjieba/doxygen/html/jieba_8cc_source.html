27template <typename Algo>
28auto Jieba::cut(std::string_view str, std::optional<size_t> max_word_length) const -> std::vector<std::string> {
29 std::u32string sentence;
30 if (not utf8_to_utf32(str, sentence)) {
31 return {};
32 }
33 size_t word_len = max_word_length.value_or(500);
34 std::optional<Algo> segmenter;
35 if constexpr (std::is_same_v<Algo, MPSegment> or std::is_same_v<Algo, FullSegment>) {
36 segmenter.emplace(trie_, word_len);
37 } else if constexpr (std::is_same_v<Algo, HMMSegment>) {
38 segmenter.emplace(hmm_);
39 } else if constexpr (std::is_same_v<Algo, MixSegment> or std::is_same_v<Algo, QuerySegment>) {
40 segmenter.emplace(hmm_, trie_, word_len);
41 } else {
42 static_assert(sizeof(Algo) == 0, "Unsupported algorithm");
43 }
44
45 std::vector<std::u32string_view> segments;
46 segments.reserve(sentence.size());
47 for (const std::u32string_view &piece : PreSegmenter(U" \t，。")(sentence)) {
48 auto segs = (*segmenter)(piece);
49 segments.insert(segments.end(), segs.begin(), segs.end());
50 }
51
52 size_t size = segments.size();
53 std::vector<std::string> ret(segments.size());
54 for (size_t i = 0; i < size; ++i) {
55 assert(utf32_to_utf8(segments[i], ret[i]));
56 }
57 segments.shrink_to_fit();
58 return ret;
59}

65auto Jieba::tag_word(std::string_view str) const -> std::string_view {
66 std::u32string encoded;
67 if (not utf8_to_utf32(str, encoded)) {
68 return Tagger::TAG_UNK;
69 }
70 auto ret = trie_.match(encoded);
71 if (ret != nullptr) {
72 return ret->tag;
73 }
74 return Tagger::classify_unknown_word(encoded);
75}

77auto Jieba::tag_sentence(std::string_view str) const -> std::vector<std::pair<std::string, std::string_view>> {
78 std::u32string sentence;
79 if (not utf8_to_utf32(str, sentence)) {
80 return {};
81 }
82 Tagger tagger(hmm_, trie_, 500);
83 std::vector<std::pair<std::string, std::string_view>> tags;
84 tags.reserve(sentence.size());
85 std::string u8word;
86
87 for (const std::u32string_view &piece : PreSegmenter(U" \t，。")(sentence)) {
88 for (const auto &[word, tag] : tagger(piece)) {
89 assert(utf32_to_utf8(word, u8word));
90 tags.emplace_back(std::move(u8word), tag);
91 }
92 }
93 tags.shrink_to_fit();
94 return tags;
95}
